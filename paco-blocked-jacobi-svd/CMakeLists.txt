cmake_minimum_required(VERSION 3.16)
project(paco_blocked_jacobi_svd LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SOC_VERSION "ascend910b3")
set(ASCEND_CANN_PACKAGE_PATH $ENV{ASCEND_HOME_PATH} CACHE PATH "CANN package")
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(blocked_fused STATIC blocked_jacobi_fused.cpp)
ascendc_compile_options(blocked_fused PRIVATE "-Wno-deprecated-declarations")
ascendc_include_directories(blocked_fused PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/include)
add_executable(jacobi_svd_main main.cpp)
target_include_directories(jacobi_svd_main PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/include)
target_link_directories(jacobi_svd_main PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(jacobi_svd_main PRIVATE blocked_fused ascendcl)
